<body>
  <main>
    <h1>Time List</h1>
    <div class="subtitle">
      Record what you are doing, one moment at a time.
    </div>

    <form id="timeForm">
      <input
        id="activityInput"
        type="text"
        placeholder="What are you doing?"
        autocomplete="off"
        required
      />

      <input
        id="timeInput"
        type="time"
        required
      />

      <button type="submit">Add</button>
    </form>

    <section id="recordList" class="list"></section>
  </main>

  <script>
    const form = document.getElementById("timeForm");
    const activityInput = document.getElementById("activityInput");
    const timeInput = document.getElementById("timeInput");
    const recordList = document.getElementById("recordList");

    let records = [];

    function getCurrentTime() {
      const now = new Date();

      const hours = String(now.getHours()).padStart(2, "0");
      const minutes = String(now.getMinutes()).padStart(2, "0");

      return `${hours}:${minutes}`;
    }

    function renderRecords() {
      recordList.innerHTML = "";

      if (records.length === 0) {
        const emptyMessage = document.createElement("div");
        emptyMessage.className = "empty";
        emptyMessage.textContent = "No records yet.";
        recordList.appendChild(emptyMessage);
        return;
      }

      records.sort((a, b) => {
        return a.time.localeCompare(b.time);
      });

      records.forEach((record, index) => {
        const recordElement = document.createElement("div");
        recordElement.className = "record";

        const timeElement = document.createElement("div");
        timeElement.className = "record-time";
        timeElement.textContent = record.time;

        const textElement = document.createElement("div");
        textElement.className = "record-text";
        textElement.textContent = record.text;

        const deleteButton = document.createElement("button");
        deleteButton.className = "delete-button";
        deleteButton.type = "button";
        deleteButton.textContent = "Delete";

        deleteButton.addEventListener("click", function () {
          records.splice(index, 1);
          renderRecords();
        });

        recordElement.appendChild(timeElement);
        recordElement.appendChild(textElement);
        recordElement.appendChild(deleteButton);

        recordList.appendChild(recordElement);
      });
    }

    timeInput.value = getCurrentTime();

    form.addEventListener("submit", function (event) {
      event.preventDefault();

      const activity = activityInput.value.trim();
      const selectedTime = timeInput.value || getCurrentTime();

      if (activity === "") {
        return;
      }

      records.push({
        text: activity,
        time: selectedTime
      });

      activityInput.value = "";
      timeInput.value = getCurrentTime();

      renderRecords();

      activityInput.focus();
    });

    renderRecords();
  </script>
